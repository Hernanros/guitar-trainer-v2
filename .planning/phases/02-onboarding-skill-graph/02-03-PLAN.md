```python
"""Onboarding parse — single Sonnet call that turns free-text song lists into
a canonicalized songs list + a 3-level DAG skill graph.

Phase 4 cost-governor interception point: this is the ONLY function that calls
Sonnet during onboarding. Do not sprinkle client.messages.create() calls anywhere else.
"""
import asyncio
import logging
from typing import Any

from anthropic import APITimeoutError, APIError

from app.ai.client import get_client, SONNET_MODEL
from app.models.skill_node import SonnetOnboardingOutput

logger = logging.getLogger(__name__)

# Fixed root taxonomy per D-08. Sonnet MUST use exactly these names.
FIXED_ROOTS = ["Rhythm", "Lead", "Chord Voicings", "Fingerstyle", "Music Theory", "Timing"]

SYSTEM_PROMPT = f"""You are a guitar-teaching assistant helping bootstrap a personalized skill graph for a new user.

You will receive three lists of songs describing what the user can play, is working on, and aspires to play.

Your job: return a structured payload with two parts.

PART 1 — SONGS: for each song the user mentioned, produce a canonicalized entry with title + artist + category + a list of skill_temp_ids that describe the techniques required.

PART 2 — SKILL_GRAPH: a 3-level tree of skill nodes.

CRITICAL RULES:
1. Root-level nodes MUST be exactly these six, no more, no fewer: {', '.join(FIXED_ROOTS)}. Use these names verbatim.
2. Sub-domain nodes (level='sub') are your choice — you decide the meaningful subdivisions under each root. Aim for 2-5 sub-domains per root that you actually populate with leaves.
3. Leaf nodes (level='leaf') represent specific playable skills. Each leaf MUST have tempo_bin_low and tempo_bin_high. tempo_bin_high MUST equal tempo_bin_low + 5 (5-bpm bins). Pick a reasonable tempo range for the skill.
4. Every leaf MUST have a parent_temp_id pointing to a sub-domain. Every sub-domain MUST have a parent_temp_id pointing to one of the six roots. Roots have parent_temp_id=null.
5. Do NOT include mastery values — mastery is always 0 at bootstrap (managed server-side, not by you).
6. Every skill_temp_id in a song's skill_temp_ids MUST reference an existing SonnetSkillNodeProposal.temp_id in your skill_graph.
7. Assign temp_ids as short unique strings within this response (e.g., "root-rhythm", "sub-shuffle", "leaf-shuffle-e-100").

If a song list is empty, produce roots only for that category's implied focus but do NOT invent songs.

If ALL THREE song lists are empty, return the six roots only with NO sub-domains or leaves — the user has provided no signal to build from."""

# Anthropic tool-use definition — Sonnet returns via this tool.
_TOOL_NAME = "emit_onboarding_output"
_TOOL_DEF = {
    "name": _TOOL_NAME,
    "description": "Emit the canonicalized songs + initial skill graph for this user.",
    "input_schema": SonnetOnboardingOutput.model_json_schema(),
}

async def run_onboarding_parse(
    raw_input: dict,
    *,
    timeout_seconds: float = 30.0,
) -> SonnetOnboardingOutput:
    """
    Single Sonnet call per onboarding (D-05). Structured tool-use output (D-06).
    Failure handling (D-07): one retry with widened timeout, then re-raise.
    Caller (POST /api/v1/users) catches and applies fail-open (bootstrap graph).

    raw_input shape: { "can_play": str, "working_on": str, "aspirational": str }
    """
    user_content = _format_user_message(raw_input)
    client = get_client()

    async def _call(timeout: float) -> SonnetOnboardingOutput:
        resp = await asyncio.wait_for(
            client.messages.create(
                model=SONNET_MODEL,
                max_tokens=4096,
                system=SYSTEM_PROMPT,
                messages=[{"role": "user", "content": user_content}],
                tools=[_TOOL_DEF],
                tool_choice={"type": "tool", "name": _TOOL_NAME},
            ),
            timeout=timeout,
        )
        # Extract tool_use block from the response
        tool_use = next((b for b in resp.content if getattr(b, "type", None) == "tool_use"), None)
        if tool_use is None:
            raise RuntimeError("Sonnet did not emit a tool_use block despite forced tool_choice.")
        # Pydantic-validate the structured output; also enforces D-08 root taxonomy at the model level
        return SonnetOnboardingOutput.model_validate(tool_use.input)

    try:
        return await _call(timeout_seconds)
    except (APITimeoutError, asyncio.TimeoutError, APIError) as e:
        logger.warning("Sonnet onboarding call failed once (%s). Retrying with widened timeout.", type(e).__name__)
        return await _call(timeout_seconds * 2)      # D-07 widened retry

def _format_user_message(raw_input: dict) -> str:
    can_play = raw_input.get("can_play", "") or "(none)"
    working = raw_input.get("working_on", "") or "(none)"
    aspire = raw_input.get("aspirational", "") or "(none)"
    return (
        f"Songs the user says they can play:\n{can_play}\n\n"
        f"Songs the user is working on:\n{working}\n\n"
        f"Songs the user aspires to play:\n{aspire}\n\n"
        f"Emit the structured onboarding output now."
    )
```

**IMPORTANT prompt-injection mitigation:** the `_format_user_message` function wraps user text with static labels. The system prompt establishes that user text is DATA not INSTRUCTIONS. This is not perfect — for the POC it is acceptable. Threat model documents this.

```python
async def _persist_bootstrap(
    db: AsyncSession,
    user_id: UUID,
    output: SonnetOnboardingOutput,
) -> list[SkillNode]:
    """Atomically persist Sonnet's structured output into skill_nodes + songs + song_skills.
    Returns the inserted SkillNode rows so the caller can build SkillGraphResponse."""
    # 1) map Sonnet temp_ids -> real UUIDs
    temp_to_uuid: dict[str, UUID] = {p.temp_id: uuid4() for p in output.skill_graph}

    # 2) coerce any invalid root names to closest known root (D-08 defense-in-depth)
    root_lookup = {r.lower(): r for r in FIXED_ROOTS}
    for p in output.skill_graph:
        if p.level == "root" and p.name not in FIXED_ROOTS:
            fallback = root_lookup.get(p.name.lower())
            if fallback:
                logger.warning("Sonnet returned non-canonical root '%s' — coerced to '%s'", p.name, fallback)
                p.name = fallback
            else:
                logger.warning("Sonnet returned unknown root '%s' — coerced to 'Music Theory'", p.name)
                p.name = "Music Theory"

    # 3) Insert skill_nodes in two passes (roots first, then subs, then leaves) to satisfy self-referential FK
    #    Actually — since parent_id is NULLABLE and all FKs check on commit (deferrable), we can insert in any order
    #    if the session is set up correctly. Safest: sort by level.
    level_order = {"root": 0, "sub": 1, "leaf": 2}
    ordered = sorted(output.skill_graph, key=lambda p: level_order[p.level])

    for prop in ordered:
        parent_uuid = temp_to_uuid.get(prop.parent_temp_id) if prop.parent_temp_id else None
        db.add(SkillNode(
            id=temp_to_uuid[prop.temp_id],
            user_id=user_id,
            name=prop.name,
            level=prop.level,
            parent_id=parent_uuid,
            tempo_bin_low=prop.tempo_bin_low,
            tempo_bin_high=prop.tempo_bin_high,
            # mastery defaults to 0.0 server_default (D-11)
        ))

    await db.flush()   # get SkillNode IDs available for song_skills

    # 4) Insert songs
    song_id_by_key: dict[tuple[str, str], int] = {}
    for song_prop in output.songs:
        song = Song(
            title=song_prop.title,
            artist=song_prop.artist,
            genre=None,
            difficulty=None,
            bpm=None,
            key=None,
            breakdown={"placeholder": "Phase 3 will populate breakdown"},   # non-null JSONB per 0001 schema
            user_id=user_id,
            category=song_prop.category,
        )
        db.add(song)
        await db.flush()
        song_id_by_key[(song_prop.title.lower(), song_prop.artist.lower())] = song.id

    # 5) Insert song_skills
    for song_prop in output.songs:
        song_id = song_id_by_key[(song_prop.title.lower(), song_prop.artist.lower())]
        for temp_id in song_prop.skill_temp_ids:
            skill_uuid = temp_to_uuid.get(temp_id)
            if skill_uuid is None:
                logger.warning("Sonnet referenced unknown temp_id '%s' in song '%s' — skipping.", temp_id, song_prop.title)
                continue
            db.add(SongSkill(song_id=song_id, skill_node_id=skill_uuid))

    await db.flush()

    # Return the freshly-created skill_nodes for the response
    result = await db.execute(select(SkillNode).where(SkillNode.user_id == user_id))
    return list(result.scalars().all())


def _make_bootstrap_graph(user_id: UUID) -> list[SkillNode]:
    """D-07 fail-open: return the 6 fixed roots only, no sub/leaf."""
    return [
        SkillNode(id=uuid4(), user_id=user_id, name=root, level="root", parent_id=None,
                  tempo_bin_low=None, tempo_bin_high=None)
        for root in FIXED_ROOTS
    ]


@router.post("/users", response_model=SkillGraphResponse, status_code=201)
async def bootstrap_user(
    body: UserBootstrapRequest,
    db: AsyncSession = Depends(get_db),
) -> SkillGraphResponse:
    """
    Bootstrap the user (D-04 + D-05 + D-06 + D-07):
    1. Upsert users row (idempotent).
    2. Call Sonnet once via run_onboarding_parse (retries once internally per D-07).
    3. Atomically persist songs + song_skills + skill_nodes.
    4. Set onboarded_at.
    On Sonnet failure after retry: save raw_input to users.raw_onboarding_text, insert 6-root
    bootstrap graph, set onboarded_at, return SkillGraphResponse(mode='bootstrap').
    """
    # Step 1: upsert user (also persists preferences + raw_input up-front so a mid-flight
    # crash still leaves the developer able to reprocess raw text later)
    now = datetime.now(timezone.utc)
    upsert = pg_insert(User).values(
        id=body.user_id,
        preferences=body.preferences.model_dump(),
        raw_onboarding_text=body.raw_input,
        # onboarded_at set AFTER graph persist succeeds, to keep the state machine clean:
        #   users row exists w/ raw_input but no onboarded_at → mid-flight
        #   users row w/ onboarded_at → complete (either full or bootstrap graph)
    ).on_conflict_do_update(
        index_elements=["id"],
        set_={
            "preferences": body.preferences.model_dump(),
            "raw_onboarding_text": body.raw_input,
        },
    )
    await db.execute(upsert)
    await db.flush()

    # Step 2: call Sonnet with fail-open guard
    mode: Literal["full", "bootstrap"] = "full"
    try:
        sonnet_output = await run_onboarding_parse(body.raw_input)
        skill_rows = await _persist_bootstrap(db, body.user_id, sonnet_output)
    except Exception as e:
        # D-07: on any failure (Sonnet retry exhausted, validation error, persist error),
        # roll back partial writes and fall back to bootstrap graph
        logger.exception("Sonnet-backed bootstrap failed for user %s — falling back to 6-root bootstrap graph.", body.user_id)
        await db.rollback()
        # Re-upsert the user (rollback erased it)
        await db.execute(upsert)
        # Persist bootstrap graph
        bootstrap_rows = _make_bootstrap_graph(body.user_id)
        for row in bootstrap_rows:
            db.add(row)
        skill_rows = bootstrap_rows
        mode = "bootstrap"

    # Step 3: mark onboarded_at
    await db.execute(
        pg_insert(User)
        .values(id=body.user_id, onboarded_at=now, preferences=body.preferences.model_dump(), raw_onboarding_text=body.raw_input)
        .on_conflict_do_update(index_elements=["id"], set_={"onboarded_at": now})
    )
    await db.commit()

    return SkillGraphResponse(
        nodes=[SkillNodeResponse.model_validate(r) for r in skill_rows],
        mode=mode,
    )


@router.get("/users/{user_id}/skill-graph", response_model=SkillGraphResponse)
async def get_skill_graph(
    user_id: UUID,
    db: AsyncSession = Depends(get_db),
) -> SkillGraphResponse:
    """Return the user's full skill graph (tree via parent_id, no edges list needed per D-09)."""
    # Verify user exists
    user_row = (await db.execute(select(User).where(User.id == user_id))).scalar_one_or_none()
    if user_row is None:
        raise HTTPException(status_code=404, detail=f"User {user_id} not found.")

    rows = (await db.execute(select(SkillNode).where(SkillNode.user_id == user_id))).scalars().all()
    return SkillGraphResponse(
        nodes=[SkillNodeResponse.model_validate(r) for r in rows],
        mode="full",   # get-endpoint always returns mode="full" — the fail-open marker is set at bootstrap time only
    )


@router.post("/users/{user_id}/re-run", response_model=SkillGraphResponse, status_code=200)
async def re_run_onboarding(
    user_id: UUID,
    body: UserBootstrapRequest,
    db: AsyncSession = Depends(get_db),
) -> SkillGraphResponse:
    """Settings re-run per <Claude's Discretion>: wipe user's songs + song_skills + skill_nodes,
    keep users row + preferences, then re-run the bootstrap flow with fresh input.
    Preserves user_id; overwrites preferences with the re-run body's preferences.
    """
    if body.user_id != user_id:
        raise HTTPException(status_code=400, detail="Body user_id must match path user_id.")

    # Verify user exists
    user_row = (await db.execute(select(User).where(User.id == user_id))).scalar_one_or_none()
    if user_row is None:
        raise HTTPException(status_code=404, detail=f"User {user_id} not found.")

    # Wipe order: song_skills first (FKs to both songs + skill_nodes), then songs (user-scoped),
    # then skill_nodes (user-scoped). System user's Sweet Home Chicago row is protected —
    # this endpoint only deletes for the given user_id, which is never the system UUID
    # in normal flow (though a maliciously-called re-run against system UUID would wipe it;
    # accept the risk per T-02-06 in 02-01's threat model).
    await db.execute(delete(SongSkill).where(SongSkill.song_id.in_(
        select(Song.id).where(Song.user_id == user_id)
    )))
    await db.execute(delete(Song).where(Song.user_id == user_id))
    await db.execute(delete(SkillNode).where(SkillNode.user_id == user_id))
    await db.flush()

    # Delegate to bootstrap_user's flow by calling the same helper
    # (bootstrap_user's implementation is inlined here to avoid re-reading the request body)
    now = datetime.now(timezone.utc)
    mode: Literal["full", "bootstrap"] = "full"
    try:
        sonnet_output = await run_onboarding_parse(body.raw_input)
        # Update preferences on the existing user row
        user_row.preferences = body.preferences.model_dump()
        user_row.raw_onboarding_text = body.raw_input
        skill_rows = await _persist_bootstrap(db, user_id, sonnet_output)
    except Exception as e:
        logger.exception("Sonnet-backed re-run failed for user %s — falling back to 6-root bootstrap graph.", user_id)
        await db.rollback()
        user_row = (await db.execute(select(User).where(User.id == user_id))).scalar_one()
        user_row.preferences = body.preferences.model_dump()
        user_row.raw_onboarding_text = body.raw_input
        bootstrap_rows = _make_bootstrap_graph(user_id)
        for row in bootstrap_rows:
            db.add(row)
        skill_rows = bootstrap_rows
        mode = "bootstrap"

    user_row.onboarded_at = now
    await db.commit()

    return SkillGraphResponse(
        nodes=[SkillNodeResponse.model_validate(r) for r in skill_rows],
        mode=mode,
    )
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user-typed free-text → Sonnet system prompt (raw_input) | User text becomes part of the Sonnet request; treated as data via SYSTEM_PROMPT framing but no true injection isolation exists |
| Sonnet response → server (structured tool-use output) | Sonnet may attempt to emit shapes outside the tool schema; Pydantic validation is the enforcement point |
| Sonnet response → Postgres (skill_nodes + songs + song_skills) | Sonnet-controlled text stored in DB (skill name, sub-domain name, leaf names); potential for injection if downstream SQL uses raw string interpolation (it doesn't — all inserts are via ORM/parameterized) |
| Server → Anthropic API | outbound network call over HTTPS; requires ANTHROPIC_API_KEY secret |
| Server → Railway Postgres | at-rest encryption + private-network TLS (Railway defaults) |
