```typescript
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import type { components } from './generated/schema';
import { apiFetch } from './apiClient';
import { getOrCreateUserId, setOnboardedAt, clearOnboardedAt, clearWizardState } from './mmkv';

export type UserResponse = components['schemas']['UserResponse'];
export type SkillGraphResponse = components['schemas']['SkillGraphResponse'];
export type UserBootstrapRequest = components['schemas']['UserBootstrapRequest'];
export type UserPreferences = components['schemas']['UserPreferences'];
export type SkillNodeResponse = components['schemas']['SkillNodeResponse'];

// ... (existing useUser + useUserBootstrap from 02-01 — DO NOT REMOVE)

export function useSkillGraph() {
  const userId = getOrCreateUserId();
  return useQuery({
    queryKey: ['skill-graph', userId],
    queryFn: () => apiFetch<SkillGraphResponse>(`/api/v1/users/${userId}/skill-graph`),
    // Skill graph rarely changes without an explicit action — long stale time OK
  });
}

export function useUserReonboard() {
  const qc = useQueryClient();
  const userId = getOrCreateUserId();
  return useMutation({
    mutationFn: (body: UserBootstrapRequest) =>
      apiFetch<SkillGraphResponse>(`/api/v1/users/${userId}/re-run`, {
        method: 'POST',
        body: JSON.stringify(body),
      }),
    onSuccess: (graph) => {
      setOnboardedAt(new Date().toISOString());
      clearWizardState();
      qc.setQueryData(['skill-graph', userId], graph);
      qc.invalidateQueries({ queryKey: ['user', userId] });
    },
  });
}
```

Note: `useUserReonboard` is NOT invoked from Settings directly in this slice. Settings' re-run flow **clears MMKV state and routes back to /onboarding** — which then triggers the full wizard again → the wizard's Complete tap already calls `useUserBootstrap`. The `useUserReonboard` hook exists for future use (e.g., a "quick re-run with same songs" feature) — for now, `POST /re-run` on the server is called via `useUserBootstrap`... wait, that would hit `POST /users`, not `POST /users/{id}/re-run`. **Correction:** the re-run flow needs to call the re-run endpoint (which wipes first), NOT the bootstrap endpoint (which does an upsert). Two options:

**Option A (chosen):** Settings triggers `clearOnboardedAt() + clearWizardState()` + `router.replace('/onboarding')`. The wizard runs. Preferences.tsx's Complete handler branches on `getOnboardedAt() != null` at wizard-start — but MMKV was cleared, so it can't. Instead: Settings sets a MMKV flag `re_run_pending = true`. Preferences.tsx checks this flag and, if true, calls `useUserReonboard.mutate` instead of `useUserBootstrap.mutate`. On success, clears the flag.

**Option B (simpler):** Settings calls `useUserReonboard.mutate` DIRECTLY with the LAST-KNOWN bootstrap payload... but Settings doesn't have the raw text. This means Option A is cleaner.

**Chosen: Option A with MMKV flag.** Update this action accordingly.
