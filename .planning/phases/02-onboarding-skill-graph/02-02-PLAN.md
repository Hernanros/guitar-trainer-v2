```typescript
export type WizardSection = 'index' | 'play' | 'working-on' | 'aspire' | 'preferences';

const WIZARD_SECTION_PREFIX = 'wizard.section.';
const WIZARD_LAST_SECTION_KEY = 'wizard.last_section';

export function setWizardSection(section: WizardSection, text: string): void {
  userMmkv.set(`${WIZARD_SECTION_PREFIX}${section}`, text);
}

export function getWizardSection(section: WizardSection): string {
  return userMmkv.getString(`${WIZARD_SECTION_PREFIX}${section}`) ?? '';
}

export function setLastSection(section: WizardSection): void {
  userMmkv.set(WIZARD_LAST_SECTION_KEY, section);
}

export function getLastSection(): WizardSection | null {
  const v = userMmkv.getString(WIZARD_LAST_SECTION_KEY);
  return (v as WizardSection | undefined) ?? null;
}

export function clearWizardState(): void {
  const sections: WizardSection[] = ['index', 'play', 'working-on', 'aspire', 'preferences'];
  sections.forEach((s) => userMmkv.remove(`${WIZARD_SECTION_PREFIX}${s}`));
  userMmkv.remove(WIZARD_LAST_SECTION_KEY);
}

// Also expose a helper for the preferences section (single object vs per-key text)
const WIZARD_PREFS_KEY = 'wizard.preferences';

export function setWizardPreferences(prefs: { session_length_min: 15 | 30 | 45 | 60 | null; retention_format: 'streak' | 'weekly_digest' | 'monthly_milestone' | null }): void {
  userMmkv.set(WIZARD_PREFS_KEY, JSON.stringify(prefs));
}

export function getWizardPreferences(): { session_length_min: 15 | 30 | 45 | 60 | null; retention_format: 'streak' | 'weekly_digest' | 'monthly_milestone' | null } {
  const raw = userMmkv.getString(WIZARD_PREFS_KEY);
  if (!raw) return { session_length_min: null, retention_format: null };
  try { return JSON.parse(raw); } catch { return { session_length_min: null, retention_format: null }; }
}
```

```typescript
import React, { useEffect, useState, useRef } from 'react';
import { View, TextInput, StyleSheet } from 'react-native';
import { setWizardSection, getWizardSection, type WizardSection } from '../api/mmkv';

interface SongInputAreaProps {
  section: WizardSection;                 // 'play' | 'working-on' | 'aspire' (index/preferences don't use this)
  placeholder: string;
  minHeight?: number;                     // aspire section may pass 200 vs default 140
  onChangeText?: (text: string) => void;  // optional parent notification for enabling Continue button
}

export function SongInputArea({ section, placeholder, minHeight = 140, onChangeText }: SongInputAreaProps) {
  const [text, setText] = useState<string>(() => getWizardSection(section));
  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
  }, []);

  const handleChange = (next: string) => {
    setText(next);
    onChangeText?.(next);
    if (debounceRef.current) clearTimeout(debounceRef.current);
    debounceRef.current = setTimeout(() => setWizardSection(section, next), 300);
  };

  return (
    <View style={[styles.card, { minHeight }]}>
      <TextInput
        style={styles.input}
        multiline
        value={text}
        onChangeText={handleChange}
        placeholder={placeholder}
        placeholderTextColor="#666"
        autoCapitalize="none"
        autoCorrect={false}
        textAlignVertical="top"
      />
    </View>
  );
}

const styles = StyleSheet.create({
  card: {
    backgroundColor: '#242424',
    borderRadius: 10,
    padding: 14,
    marginBottom: 16,
    borderLeftWidth: 3,
    borderLeftColor: '#E07B39',
  },
  input: {
    color: '#F5F5F5',
    fontSize: 16,
    lineHeight: 22,
    minHeight: 120,
  },
});
```

```typescript
import React, { useState } from 'react';
import { View, Text, StyleSheet, ActivityIndicator } from 'react-native';
import { router } from 'expo-router';
import { FletcherIntroCard } from '../../components/FletcherIntroCard';
import { SessionLengthChips } from '../../components/SessionLengthChips';
import { RetentionFormatRadio } from '../../components/RetentionFormatRadio';
import { useUserBootstrap, type UserBootstrapRequest } from '../../api/users';
import {
  getOrCreateUserId,
  getWizardSection,
  clearWizardState,
  setWizardPreferences,
  getWizardPreferences,
  setLastSection,
} from '../../api/mmkv';

type LengthValue = 15 | 30 | 45 | 60;
type RetentionFormat = 'streak' | 'weekly_digest' | 'monthly_milestone';

export default function OnboardingPreferences() {
  const initial = getWizardPreferences();
  const [sessionLength, setSessionLength] = useState<LengthValue | null>(initial.session_length_min);
  const [retention, setRetention] = useState<RetentionFormat | null>(initial.retention_format);
  const bootstrap = useUserBootstrap();

  React.useEffect(() => {
    setWizardPreferences({ session_length_min: sessionLength, retention_format: retention });
  }, [sessionLength, retention]);

  const onComplete = () => {
    const parseLines = (raw: string) => raw.split('\n').map((s) => s.trim()).filter(Boolean);
    const rawCanPlay = getWizardSection('play');
    const rawWorking = getWizardSection('working-on');
    const rawAspire = getWizardSection('aspire');
    const body: UserBootstrapRequest = {
      user_id: getOrCreateUserId(),
      songs: {
        can_play: parseLines(rawCanPlay),
        working_on: parseLines(rawWorking),
        aspirational: parseLines(rawAspire),
      },
      preferences: {
        session_length_min: sessionLength!,             // guarded below via disabled
        // D-13: default to streak if unpicked
        retention_format: retention ?? 'streak',
      },
      raw_input: {
        can_play: rawCanPlay,
        working_on: rawWorking,
        aspirational: rawAspire,
      },
    };
    bootstrap.mutate(body, {
      onSuccess: () => {
        clearWizardState();
        setLastSection('index');   // reset for potential future re-run
        router.replace('/(tabs)');
      },
    });
  };

  const canComplete = sessionLength !== null;   // D-12 forces intentional choice; retention has a default

  if (bootstrap.isPending) {
    return (
      <View style={styles.loader}>
        <ActivityIndicator size="large" color="#E07B39" />
        <Text style={styles.loaderText}>Fletcher is listening...</Text>
      </View>
    );
  }

  return (
    <FletcherIntroCard
      heading="How long do you have most days?"
      body="Pick a session length. Fletcher will size lessons to fit."
      cta="Complete"
      onNext={onComplete}
      disabled={!canComplete}
      progress={{ current: 5, total: 5 }}
    >
      <SessionLengthChips value={sessionLength} onChange={setSessionLength} />
      <Text style={styles.subHeading}>How should Fletcher mark the wins?</Text>
      <RetentionFormatRadio value={retention} onChange={setRetention} />
      {bootstrap.isError && (
        <View style={styles.errorBox}>
          <Text style={styles.errorText}>Fletcher lost the thread. Try that again.</Text>
        </View>
      )}
    </FletcherIntroCard>
  );
}

const styles = StyleSheet.create({
  loader: { flex: 1, backgroundColor: '#1A1A1A', alignItems: 'center', justifyContent: 'center' },
  loaderText: { color: '#F5F5F5', marginTop: 12, fontSize: 14 },
  subHeading: { color: '#F5F5F5', fontSize: 16, fontWeight: '700', marginTop: 8, marginBottom: 12 },
  errorBox: { backgroundColor: '#3A1F1F', padding: 12, borderRadius: 6, marginBottom: 12, borderLeftWidth: 3, borderLeftColor: '#ff6b6b' },
  errorText: { color: '#ffb0b0', fontSize: 14 },
});
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user-typed free-text → MMKV (device-local) | Song titles + arbitrary user text buffered in MMKV wizard slots |
| user-typed free-text → POST /api/v1/users (raw_input JSONB) | Free-text sent to server, stored in `users.raw_onboarding_text` |
| mobile client → server (X-User-ID) | Existing boundary from 02-01; unchanged this slice |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-02-01 | Information Disclosure | MMKV wizard-state slots persisting typed text | accept | MMKV is device-local; typed content is song titles + free-form user text. Not encrypted at rest beyond iOS/Android disk encryption. POC scope per PROJECT.md single-user. Wizard state IS cleared on successful Complete (Task 2 calls `clearWizardState()` in `preferences.tsx` onSuccess) so abandoned data does not linger. |
| T-02-02-02 | Information Disclosure | `raw_input` field posted to `POST /api/v1/users` | mitigate | The endpoint receives verbatim user text and stores it in `users.raw_onboarding_text` JSONB. Threat: this can contain PII if the user types names ("jamming with my brother on Little Wing"). Mitigations for POC: (a) single-user POC per PROJECT.md — no cross-user exposure risk; (b) transport uses HTTPS (Railway default); (c) at-rest encryption via Railway Postgres. Documented deferral: PII redaction of raw_input becomes a Phase 4+ concern when multi-user lands. This slice does NOT send raw_input to Sonnet (02-03 does). |
| T-02-02-03 | Denial of Service | Complete-tap mutation with pathologically-large text input | mitigate | Bootstrap endpoint (02-01 stub) is idempotent + cheap (upsert). The stub does not process the text. Size ceiling: React Native TextInput does not enforce a client-side length cap. **Add a soft cap** in `SongInputArea`: reject input longer than 4000 chars per section (silently truncate on write to MMKV — user sees the visible length). Server-side FastAPI will happily accept larger payloads but this stub's DB write is small. 02-03's Sonnet call is where real DoS risk lands — its threat model addresses cost caps. |
| T-02-02-04 | Tampering | User modifies wizard state via MMKV file access (jailbroken device) | accept | POC scope. Single-user POC. The user is the developer. If jailbroken user modifies their own MMKV to send different data to bootstrap, the worst case is a corrupted local skill graph — which they can already re-run from Settings (02-04). No cross-user impact. |
| T-02-02-SC | Tampering | Package installs | mitigate | This slice adds no new dependencies (all UI is built from existing `react`, `react-native`, `expo-router` per package.json). If executor discovers a need for a chip/radio library during implementation, they MUST STOP and add a slopcheck task before installing. Default policy: hand-roll all components (which is what Task 1 does). |
