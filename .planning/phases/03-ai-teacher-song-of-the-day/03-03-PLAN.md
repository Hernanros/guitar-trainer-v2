This slice closes the daily loop: Slice A picks the song, Slice B teaches it, Slice C reads the user's honest self-report and writes back to the graph — deterministically, no LLM.

Purpose: This is the final vertical slice that makes Phase 3's user story true end-to-end. Without Slice C, the daily loop does not close — tomorrow's song can't reflect yesterday's rating because no rating exists. This slice also cements the "deterministic writes to skill graph from self-report only" project principle (PROJECT.md) — no Sonnet call is made in the write path.

Output: User taps a rating pill → mastery on all attached skill_nodes shifts → tomorrow's selector reads updated mastery → tomorrow's Song of the Day reflects yesterday's rating. Daily loop closes.
</objective>

    Create `server/app/api/v1/sessions.py` per PATTERNS.md lines 455-483 and RESEARCH §5:
    ```python
    import logging
    import uuid
    from datetime import datetime, timezone
    from decimal import Decimal
    from uuid import UUID

    from fastapi import APIRouter, Depends, HTTPException
    from sqlalchemy import func, select, text, update
    from sqlalchemy.exc import IntegrityError
    from sqlalchemy.ext.asyncio import AsyncSession

    from app.api.deps import get_tz_offset_minutes, get_user_id
    from app.db.session import get_db
    from app.models.db import SkillNode, Song, SongSkill, UserSession
    from app.models.session import SessionCreate, SessionResponse

    logger = logging.getLogger(__name__)
    router = APIRouter()

    RATING_SHIFTS: dict[str, Decimal] = {
        "not_my_tempo": Decimal("-0.05"),
        "getting_closer": Decimal("0.05"),
        "thats_what_im_looking_for": Decimal("0.15"),
    }


    @router.post("/sessions", response_model=SessionResponse, status_code=201)
    async def submit_rating(
        body: SessionCreate,
        user_id: UUID = Depends(get_user_id),
        tz_offset_minutes: int = Depends(get_tz_offset_minutes),
        db: AsyncSession = Depends(get_db),
    ) -> SessionResponse:
        # 1. Compute local calendar day server-side (never trust client)
        local_day = await db.scalar(
            text("SELECT DATE((now() AT TIME ZONE 'UTC') + (:tz * INTERVAL '1 minute'))"),
            {"tz": tz_offset_minutes},
        )

        # 2. Access control — T-03-03-01
        song = (
            await db.execute(select(Song).where(Song.id == body.song_id, Song.user_id == user_id))
        ).scalar_one_or_none()
        if song is None:
            raise HTTPException(status_code=404, detail="Song not found or not owned by user.")

        # 3. Application-level idempotency guard (legibility) — T-03-03-02 layer (a)
        already = await db.scalar(
            select(func.count(UserSession.id)).where(
                UserSession.user_id == user_id,
                UserSession.song_id == body.song_id,
                UserSession.local_calendar_day == local_day,
                UserSession.is_reroll_marker == False,  # noqa: E712 — SQLAlchemy comparison
            )
        )
        if already:
            raise HTTPException(status_code=409, detail="Already rated this song today.")

        shift = RATING_SHIFTS[body.rating]

        # 4. Single transaction — INSERT session + UPDATE mastery
        session_row = UserSession(
            id=uuid.uuid4(),
            user_id=user_id,
            song_id=body.song_id,
            rating=body.rating,
            local_calendar_day=local_day,
            tz_offset_minutes=tz_offset_minutes,
            is_reroll_marker=False,
        )
        try:
            async with db.begin():
                db.add(session_row)

                # UPDATE mastery on every leaf skill_node in song_skills for this song.
                # T-03-03-03: SQL-side clamp with typed numeric literals per RESEARCH §8 landmine 11.
                # T-03-07 (RESEARCH §9 Q5): explicit updated_at=func.now() bump — do NOT rely on
                #   SQLAlchemy onupdate firing on bulk UPDATE (it does not, per Q5).
                # T-03-03-07: NO reference to SongSkill.weight — D-07 equal-weight.
                await db.execute(
                    update(SkillNode)
                    .where(
                        SkillNode.id.in_(
                            select(SongSkill.skill_node_id).where(SongSkill.song_id == body.song_id)
                        ),
                        SkillNode.user_id == user_id,  # T-03-03-01 defense in depth
                    )
                    .values(
                        mastery=text("LEAST(1.0::numeric, GREATEST(0.0::numeric, mastery + :shift::numeric))").bindparams(shift=shift),
                        updated_at=func.now(),
                    )
                )
        except IntegrityError as exc:
            # T-03-03-02 layer (b): DB partial-unique index caught race — return 409
            if "uq_user_sessions_daily_rating" in str(exc):
                raise HTTPException(status_code=409, detail="Already rated this song today.")
            raise

        await db.refresh(session_row)
        return SessionResponse(
            id=session_row.id,
            user_id=session_row.user_id,
            song_id=session_row.song_id,
            rating=session_row.rating.value if hasattr(session_row.rating, "value") else session_row.rating,
            local_calendar_day=local_day.isoformat(),
            rated_at=session_row.rated_at.isoformat(),
        )
    ```

    import { useMutation, useQueryClient } from '@tanstack/react-query';
    import type { components } from './generated/schema';
    import { apiFetch } from './apiClient';
    import { getOrCreateUserId } from './mmkv';
    import { localCalendarDay } from './todaySong';

    export type RatingLiteral = 'not_my_tempo' | 'getting_closer' | 'thats_what_im_looking_for';
    export type SessionResponse = components['schemas']['SessionResponse'];
    // Revision A: import TodaySongResponse so onSuccess can patch the today-song cache.
    export type TodaySongResponse = components['schemas']['TodaySongResponse'];

    export function useSubmitRating() {
      const qc = useQueryClient();
      const userId = getOrCreateUserId();
      return useMutation({
        mutationFn: (body: { song_id: number; rating: RatingLiteral }) =>
          apiFetch<SessionResponse>('/api/v1/sessions', {
            method: 'POST',
            body: JSON.stringify(body),
          }),
        onSuccess: (response, variables) => {
          // Revision A (BLOCKER 1 fix): patch the today-song cache with the freshly-rated payload.
          // Server determinism (D-01/D-09) guarantees the deterministic pick does not change within the day,
          // so patching is safe. We patch (not invalidate) to avoid an extra network round-trip; the cache
          // simply gains its `rated` field, and SongOfDayCard on Today re-renders as the already-rated variant.
          const localDay = localCalendarDay();
          qc.setQueryData<TodaySongResponse>(
            ['today-song', userId, localDay],
            (old) => old
              ? { ...old, rated: { rating: variables.rating, rated_at: response.rated_at } }
              : old,
          );
          // Skill graph mastery changed → invalidate so tomorrow's Today song reflects yesterday's rating.
          qc.invalidateQueries({ queryKey: ['skill-graph', userId] });
          // D-11: breakdown cache-forever. Do NOT invalidate breakdown.
          // Revision A: today-song is PATCHED (not invalidated) so the deterministic pick stays stable
          //   AND the client immediately reflects the rated state without a second GET.
        },
        onError: async (err) => {
          const msg = err instanceof Error ? err.message : String(err);
          if (msg.includes('HTTP 409')) {
            // UI-SPEC §10: 409 is state-sync, not error. Invalidate today-song so `rated` field populates
            //   from the server (covers the case where the app missed the initial success write).
            await qc.invalidateQueries({ queryKey: ['today-song', userId, localCalendarDay()] });
          }
          // Other errors propagate to caller's error state.
        },
      });
    }
